function toggleAccordion(id) {
  const card = document.querySelector(`[data-accordion="${id}"]`);
  const icon = card.querySelector('.accordion-icon');
  card.classList.toggle('accordion-open');
  icon.style.transform = card.classList.contains('accordion-open') ? 'rotate(180deg)' : 'rotate(0)';
}

function copyText(text, btn) {
  navigator.clipboard.writeText(text).then(() => {
    btn.textContent = '已复制';
    setTimeout(() => { btn.textContent = '复制'; }, 2000);
  });
}
